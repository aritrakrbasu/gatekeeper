# Plan: RepoDetails.jsx API Integration

## Context

The `RepoDetails.jsx` component currently uses hardcoded mock data for repositories, merge requests, policies, and health metrics. This plan connects the component to real backend APIs to display live data from GitLab and the Gatekeeper system.

**Data Flow Architecture**:
1. GitLab webhook creates MR entries in MongoDB when MRs are raised
2. Auto-fix MRs are linked to their original MRs in MongoDB
3. Frontend fetches MR list from MongoDB (not GitLab directly)
4. MR data indicates whether it's an auto-fix and links to related MRs

**Critical Bug Found**: The `GET /api/repositories/{repo_id}` endpoint in `main.py` (lines 316-340) is a duplicate of the `/api/repositories/available` route and doesn't use the `repo_id` parameter. This must be fixed.

#### 2.3 Add Auto-Fix Linking Function

```python
async def link_auto_fix_to_original_mr(
    repo_id: str,
    original_mr_iid: int,
    auto_fix_mr_iid: int,
    auto_fix_branch: str
) -> dict:
    """
    Link an auto-fix MR to its original MR in MongoDB.

    When Gatekeeper generates an auto-fix MR:
    1. Mark the auto-fix MR with is_auto_fix=True and auto_fix_for=original_mr_iid
    2. Add the auto-fix MR's IID to the original MR's auto_fix_mrs list
    """
    # Update auto-fix MR document
    await mongo_client.update_one(
        collection_name=MRS_COLLECTION,
        filter={
            "repository_id": repo_id,
            "gitlab_mr_id": auto_fix_mr_iid
        },
        update={
            "$set": {
                "is_auto_fix": True,
                "auto_fix_for": original_mr_iid,
                "auto_fix_status": "applied"
            }
        }
    )

    # Update original MR document
    await mongo_client.update_one(
        collection_name=MRS_COLLECTION,
        filter={
            "repository_id": repo_id,
            "gitlab_mr_id": original_mr_iid
        },
        update={
            "$push": {"auto_fix_mrs": auto_fix_mr_iid}
        }
    )

    return {"status": "linked", "original_mr": original_mr_iid, "auto_fix_mr": auto_fix_mr_iid}
```

#### 2.4 Fetch MRs from MongoDB (Not GitLab)

```python
async def get_repository_mrs_from_db(
    repo_id: str,
    limit: int = 50,
    status_filter: Optional[str] = None
) -> List[Dict]:
    """
    Get merge requests for a repository from MongoDB.

    This replaces the GitLab API fetch with MongoDB query,
    ensuring all AI analysis and auto-fix data is included.
    """
    filter_query = {"repository_id": repo_id}

    if status_filter:
        filter_query["status"] = status_filter

    # Fetch from MongoDB
    documents = await mongo_client.find(
        collection_name=MRS_COLLECTION,
        filter=filter_query,
        limit=limit,
        sort=[("created_at", -1)]  # Newest first
    )

    # Format for frontend response
    result = []
    for doc in documents:
        result.append({
            "id": f"#{doc.get('gitlab_mr_id')}",
            "gitlab_mr_id": doc.get("gitlab_mr_id"),
            "iid": doc.get("gitlab_mr_id"),
            "title": doc.get("title"),
            "repository_id": doc.get("repository_id"),
            "repository_name": doc.get("repository_name"),
            "status": doc.get("status"),
            "author": doc.get("author"),
            "branch": doc.get("source_branch"),
            "target": doc.get("target_branch"),
            "web_url": doc.get("web_url"),
            "created_at": doc.get("created_at"),
            "updated_at": doc.get("updated_at"),
            "time": _relative_time(doc.get("created_at")),
            "changes": {
                "added": doc.get("changes", {}).get("additions", 0),
                "removed": doc.get("changes", {}).get("deletions", 0),
                "files": doc.get("changes", {}).get("files_changed", 0)
            },
            # AI Analysis Data
            "aiSummary": doc.get("ai_summary"),
            "aiIssues": doc.get("ai_issues", []),
            "healthScore": doc.get("health_score"),
            "analysisStatus": doc.get("analysis_status"),

            # Auto-Fix Data
            "isAutoFix": doc.get("is_auto_fix", False),
            "autoFixFor": doc.get("auto_fix_for"),
            "autoFixMrs": doc.get("auto_fix_mrs", []),
            "autoFixStatus": doc.get("auto_fix_status"),
        })

    return result
```
